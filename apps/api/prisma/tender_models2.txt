
model TenderQualification {
  id             String                    @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  organizationId String                    @db.Uuid
  tenderId       String                    @db.Uuid
  criterion      String
  requirement    String?
  weight         Decimal                   @default(0) @db.Decimal(5, 2)
  companyStatus  TenderQualificationStatus @default(PENDING)
  evidence       String?
  remarks        String?
  verifiedBy     String?                   @db.Uuid
  verifiedAt     DateTime?
  createdAt      DateTime                  @default(now())
  updatedAt      DateTime                  @updatedAt

  tender Tender @relation(fields: [tenderId], references: [id], onDelete: Cascade)

  @@index([tenderId])
}

model TenderRisk {
  id             String         @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  organizationId String         @db.Uuid
  tenderId       String         @db.Uuid
  risk           String
  category       String?
  level          TenderRiskLevel @default(MEDIUM)
  probability    String?
  impact         String?
  mitigation     String?
  contingency    Decimal        @default(0) @db.Decimal(18, 2)
  ownerId        String?        @db.Uuid
  dueDate        DateTime?
  status         String?        @default("OPEN")
  createdAt      DateTime       @default(now())
  updatedAt      DateTime       @updatedAt

  tender Tender @relation(fields: [tenderId], references: [id], onDelete: Cascade)

  @@index([tenderId])
}

  tender Tender @relation(fields: [tenderId], references: [id], onDelete: Cascade)

  @@index([tenderId, sequence])
